-- ================= users 用户表 =================
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '用户ID（主键，自增）',
    username VARCHAR(50) NOT NULL UNIQUE COMMENT '用户名（唯一）',
    password_hash VARCHAR(255) NOT NULL COMMENT '密码（加密存储）',
    is_admin BOOLEAN DEFAULT FALSE COMMENT '是否为管理员（0=普通用户，1=管理员）',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
) COMMENT='用户表';


-- ================= videos 视频表 =================
CREATE TABLE videos (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '视频ID（主键，自增）',
    title VARCHAR(100) NOT NULL COMMENT '视频标题',
    url VARCHAR(255) NOT NULL COMMENT '视频文件路径',
    description TEXT COMMENT '视频简介',
    user_id INT NOT NULL COMMENT '上传用户ID',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '上传时间',
    FOREIGN KEY (user_id) REFERENCES users(id)
) COMMENT='视频表';


-- ================= comments 评论表 =================
CREATE TABLE comments (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '评论ID（主键，自增）',
    video_id INT NOT NULL COMMENT '所属视频ID',
    user_id INT NOT NULL COMMENT '评论用户ID',
    content TEXT NOT NULL COMMENT '评论内容',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '评论时间',

    -- 外键约束
    CONSTRAINT fk_comment_video FOREIGN KEY (video_id) REFERENCES videos(id),
    CONSTRAINT fk_comment_user FOREIGN KEY (user_id) REFERENCES users(id)
) COMMENT='评论表';


-- ================= likes 点赞表 =================
CREATE TABLE likes (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '点赞ID（主键，自增）',
    user_id INT NOT NULL COMMENT '点赞用户ID',
    target_type VARCHAR(20) NOT NULL COMMENT '点赞对象类型（video/comment）',
    target_id INT NOT NULL COMMENT '目标ID（视频ID或评论ID）',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '点赞时间',

    -- 防止重复点赞
    UNIQUE KEY unique_like (user_id, target_type, target_id),

    FOREIGN KEY (user_id) REFERENCES users(id)
) COMMENT='点赞表（支持视频和评论）';


-- ================= follows 关注表 =================
CREATE TABLE follows (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '关注ID（主键，自增）',
    follower_id INT NOT NULL COMMENT '关注者ID',
    following_id INT NOT NULL COMMENT '被关注者ID',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '关注时间',

    -- 防止重复关注
    UNIQUE KEY unique_follow (follower_id, following_id),

    FOREIGN KEY (follower_id) REFERENCES users(id),
    FOREIGN KEY (following_id) REFERENCES users(id)
) COMMENT='用户关注关系表';


-- ================= tokens 登录令牌表 =================
CREATE TABLE tokens (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Token ID（主键，自增）',
    user_id INT NOT NULL COMMENT '所属用户ID',
    token VARCHAR(255) NOT NULL COMMENT '登录令牌（用于身份认证）',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',

    FOREIGN KEY (user_id) REFERENCES users(id)
) COMMENT='用户登录Token表';
