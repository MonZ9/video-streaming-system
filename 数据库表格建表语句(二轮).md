-- 用户表
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '用户ID',
    username VARCHAR(50) NOT NULL UNIQUE COMMENT '用户名',
    password_hash VARCHAR(255) NOT NULL COMMENT '加密后的密码',
    salt VARCHAR(64) NOT NULL COMMENT '密码盐值',
    bio TEXT COMMENT '个人简介',
    is_admin BOOLEAN DEFAULT FALSE COMMENT '是否管理员（冗余字段，RBAC 兼容）',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '注册时间'
) COMMENT='用户表';

-- 视频表
CREATE TABLE videos (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '视频ID',
    title VARCHAR(100) NOT NULL COMMENT '视频标题',
    url VARCHAR(255) NOT NULL COMMENT '视频文件路径',
    description TEXT COMMENT '视频简介',
    user_id INT NOT NULL COMMENT '上传用户ID',
    category VARCHAR(20) DEFAULT '未分类' COMMENT '视频分区',
    tags VARCHAR(500) DEFAULT '' COMMENT '用户自定义标签（逗号分隔）',
    coupon_id INT NULL COMMENT '关联的优惠券活动ID',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '上传时间',
    FOREIGN KEY (user_id) REFERENCES users(id)
) COMMENT='视频表';

-- 动态表
CREATE TABLE posts (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '动态ID',
    user_id INT NOT NULL COMMENT '发布用户ID',
    content TEXT NOT NULL COMMENT '动态内容',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '发布时间',
    FOREIGN KEY (user_id) REFERENCES users(id)
) COMMENT='用户动态表';

-- 评论表（支持视频和动态）
CREATE TABLE comments (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '评论ID',
    video_id INT NULL COMMENT '视频ID（动态评论时为空）',
    user_id INT NOT NULL COMMENT '评论用户ID',
    content TEXT NOT NULL COMMENT '评论内容',
    target_type VARCHAR(20) DEFAULT 'video' COMMENT '评论目标类型（video/post）',
    target_id INT DEFAULT 0 COMMENT '目标ID（视频ID或动态ID）',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '评论时间',
    FOREIGN KEY (user_id) REFERENCES users(id)
) COMMENT='评论表';

-- 点赞表
CREATE TABLE likes (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '点赞ID',
    user_id INT NOT NULL COMMENT '点赞用户ID',
    target_type VARCHAR(20) NOT NULL COMMENT '点赞目标类型（video/comment）',
    target_id INT NOT NULL COMMENT '目标ID',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '点赞时间',
    UNIQUE KEY unique_like (user_id, target_type, target_id),
    FOREIGN KEY (user_id) REFERENCES users(id)
) COMMENT='点赞表';

-- 关注表
CREATE TABLE follows (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '关注ID',
    follower_id INT NOT NULL COMMENT '关注者ID',
    following_id INT NOT NULL COMMENT '被关注者ID',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '关注时间',
    UNIQUE KEY unique_follow (follower_id, following_id),
    FOREIGN KEY (follower_id) REFERENCES users(id),
    FOREIGN KEY (following_id) REFERENCES users(id)
) COMMENT='用户关注关系表';

-- 收藏表
CREATE TABLE favorites (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '收藏ID',
    user_id INT NOT NULL COMMENT '用户ID',
    target_type VARCHAR(20) NOT NULL COMMENT '收藏类型（video/post）',
    target_id INT NOT NULL COMMENT '目标ID',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '收藏时间',
    UNIQUE KEY unique_fav (user_id, target_type, target_id),
    FOREIGN KEY (user_id) REFERENCES users(id)
) COMMENT='收藏表';

-- 角色表（RBAC）
CREATE TABLE roles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) UNIQUE NOT NULL
) COMMENT='角色表';

-- 权限表
CREATE TABLE permissions (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) UNIQUE NOT NULL
) COMMENT='权限表';

-- 用户角色关联表
CREATE TABLE user_roles (
    user_id INT,
    role_id INT,
    PRIMARY KEY (user_id, role_id),
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (role_id) REFERENCES roles(id)
) COMMENT='用户角色关联表';

-- 角色权限关联表
CREATE TABLE role_permissions (
    role_id INT,
    permission_id INT,
    PRIMARY KEY (role_id, permission_id),
    FOREIGN KEY (role_id) REFERENCES roles(id),
    FOREIGN KEY (permission_id) REFERENCES permissions(id)
) COMMENT='角色权限关联表';

-- 优惠券活动表
CREATE TABLE coupons (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '优惠券ID',
    title VARCHAR(100) NOT NULL COMMENT '活动标题',
    stock INT NOT NULL COMMENT '总库存',
    remain INT NOT NULL COMMENT '剩余库存',
    start_time TIMESTAMP NOT NULL COMMENT '开始时间',
    end_time TIMESTAMP NOT NULL COMMENT '结束时间',
    status INT DEFAULT 1 COMMENT '状态：1=未开始 2=进行中 3=已结束',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
) COMMENT='优惠券活动表';

-- 优惠券抢购记录表
CREATE TABLE coupon_records (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '记录ID',
    coupon_id INT NOT NULL COMMENT '优惠券ID',
    user_id INT NOT NULL COMMENT '用户ID',
    code VARCHAR(64) NOT NULL COMMENT '券码',
    grab_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '抢购时间',
    FOREIGN KEY (coupon_id) REFERENCES coupons(id),
    FOREIGN KEY (user_id) REFERENCES users(id)
) COMMENT='优惠券抢购记录表';

-- 管理员申请表
CREATE TABLE admin_apply (
    id INT AUTO_INCREMENT PRIMARY KEY COMMENT '申请ID',
    user_id INT NOT NULL COMMENT '用户ID',
    status INT DEFAULT 0 COMMENT '0=待审核 1=通过 2=拒绝',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '申请时间'
) COMMENT='管理员申请表';

-- ================= 索引优化 =================
-- 视频按用户查询、按时间排序
CREATE INDEX idx_videos_user_time ON videos(user_id, created_at DESC);
-- 动态按用户+时间查询（Feed 流常用）
CREATE INDEX idx_posts_user_time ON posts(user_id, created_at DESC);
-- 关注关系双向查询
CREATE INDEX idx_follows_follower ON follows(follower_id);
CREATE INDEX idx_follows_following ON follows(following_id);
-- 评论按目标查询
CREATE INDEX idx_comments_target ON comments(target_type, target_id);
-- 优惠券状态筛选
CREATE INDEX idx_coupons_status ON coupons(status);
-- 抢购记录按券和用户去重查询
CREATE INDEX idx_records_coupon_user ON coupon_records(coupon_id, user_id);
